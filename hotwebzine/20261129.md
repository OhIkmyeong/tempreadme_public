```js
import { get_page_info } from "../model/getPageInfo.js";
import { is_kakao_browser } from "./common/CommonSetting.js";
import { MnbCtrl } from "./components/MnbCtrl.js";
import { MainPageCtrl } from "./MainPageCtrl.js";
import { SubPageCtrl } from "./SubPageCtrl.js";
import { SmoothScroll } from "./common/SmoothScroll.js";
import { ModalCtrl } from "./components/ModalCtrl.js";
import { GnbCtrl } from "./components/GnbCtrl.js";

/**
 * common controller
 */
document.addEventListener("DOMContentLoaded", async () => {
    /* aos */
    AOS.init({ once: true });

    /* 공통:GNB */
    new GnbCtrl().init();

    /* 공통:MNB */
    new MnbCtrl().init();

    /* 공통:모달컨트롤러 */
    new ModalCtrl().init();

    /* 공통:부드러운 스크롤 */
    new SmoothScroll().init();

    /* 페이지별 처리 */
    const pageInfo = get_page_info();
    console.log(pageInfo);
    switch (pageInfo?.type) {
        case "main": {
            /* 메인페이지인 경우 */
            new MainPageCtrl().init();
        } break;
        case "sub": {
            /* 서브페이지인 경우 */
            await new SubPageCtrl()
                .set_page_info(pageInfo)
                .init();
        } break;
        default: {
            console.error("no pageInfo type".pageInfo?.type);
        } break;
    }//switch

    /* 카카오 인앱브라우저 처리 */
    is_kakao_browser();
});
```
