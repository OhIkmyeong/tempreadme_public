export class ScheduleCtrl {
    /**
     * 스케쥴러 작동을 시작합니다...
     */
    static async init() {
        const { year, month, date } = DateParser.date_to_string(new Date());

    /**

        /* 타임테이블 */
        await ScheduleCtrl.refill_timetable({
            targetId: timetableId,
            year,
            month,
            date
        });
        /* 달력 */
        await ScheduleCtrl.refill_calendar({
            targetId: calendarId,
            year,
            month,
            date
        })

        ScheduleCtrl.add_on_click_calendar({
            timetableId,
            calendarId
        });
    }//reset_dom_whole

    /**
     * 타임 테이블 채우고 이벤트 더함

    }//refill_timetable

     * 달력 채우고 이벤트 더함
     * @param {*} param0 
    */
    static refill_calendar = async ({ targetId, year, month, day }) => {
        const timetables = await ScheduleModel.get_monthly_timetable({ year, month });
        MainScheduleView.fill_calendar({ targetId, year, month, timetables });
    }//refill_calendar

    /**
     * month는 String으로 받음
     * @param {*} param0 
     */
    static async set_remember_calendar({ calendarId, year, month, date }) {
        const timeout = 5000;
        if (!calendarId) throw new Error("calendarId는 필수");
        if (!year || !month || !date) throw new Error("year,month,date 다 string값으로 넣어주세요");

        return new Promise((res, rej) => {
            // 이미 DOM에 있으면 바로 처리
            const $existing = document.getElementById(calendarId);
            if ($existing) {
                $existing.dataset.dateRemember = `${year}-${month}-${date}`;
                return res("calendar에 curr date 설정 완료");
            }

            // 타임아웃 설정
            const timer = setTimeout(() => {
                observer.disconnect();
                rej(new Error(`#${calendarId} $calendar DOM 못 찾음 (timeout: ${timeout}ms)`));
            }, timeout);

            // DOM 추가 감지
            const observer = new MutationObserver(() => {
                const $calendar = document.getElementById(calendarId);
                if (!$calendar) return;

                observer.disconnect();
                clearTimeout(timer);
                $calendar.dataset.dateRemember = `${year}-${month}-${date}`;
                res("calendar에 curr date 설정 완료");
            });

            observer.observe(document.body, { childList: true, subtree: true });
        });
    }

    /**
     * 
     * @param {*} param0 
     */
    static add_on_click_calendar = ({ timetableId, calendarId }) => {
        const $calendar = document.getElementById(calendarId);
        if (!$calendar) { throw new Error("DOM 못 찾음"); }
        $calendar.addEventListener("click", async (e) => {
            /* 이전, 다음 버튼인경우 */
            if (e.target.classList.contains("calendar-btn-month")) {
                const { dateRough } = $calendar.dataset;
                if (!dateRough) { throw new Error("dateRough값이 없음"); }
                const { year, month, date } = DateParser.string_to_object(dateRough);
                const offset = e.target.classList.contains("prev") ? -2 : 0;
                const dateString = DateParser.date_to_string(new Date(year, month + offset, 1));
                const dateTodayString = DateParser.date_to_string(new Date());
                const isTodayMonth = (dateTodayString.year === dateString.year && dateTodayString.month === dateString.month);
                /* 달력만 새로 그려 */
                await ScheduleCtrl.refill_calendar({
                    targetId: calendarId,
                    year: dateString.year,
                    month: dateString.month
                });
                /* 클릭 이벤트 더하기 */
                ScheduleCtrl.add_on_click_calendar({
                    timetableId,
                    calendarId
                });
                return;
            }//prevNext

            /* 날짜버튼 클릭한 경우 */
            if (e.target.classList.contains("calendar-btn-date")) {
                const { date: dateString } = e.target.dataset;
                if (!dateString) {
                    throw new Error("버튼에 dataset date 정보 없음");
                }
                const [year, month, date] = dateString.split("-");
                /* 선택한 날짜 기억하기 */
                await ScheduleCtrl.set_remember_calendar({
                    calendarId,
                    year,
                    month,
                    date
                });
                /* 타임 테이블만 reset */
                await ScheduleCtrl.refill_timetable({
                    targetId: timetableId,
                    year,
                    month,
                    date
                });
                /*  해당 버튼에 on 표시 */

                /* 해당 위치로 스크롤 */

export class DateParser {
    /**
     * 
     * @param {String} dateStr "YYYY-MM-DD"
     * @returns {String} "YYYY.MM.DD" 
     */
    static dash_to_dot(dateStr) {
        const d = new Date(dateStr);

        const year = d.getFullYear();
        const month = String(d.getMonth() + 1).padStart(2, '0');
        const date = String(d.getDate()).padStart(2, '0');

        return `${year}.${month}.${date}`;
    }//dash_to_dot

    /**
     * Date 객체로부터
     * @param {Date} d 
     * @returns {Object}
     */
    static date_to_string(d) {
        if (!d) { d = new Date(); }
        const year = d.getFullYear();
        const month = String(d.getMonth() + 1).padStart(2, '0');
        const date = String(d.getDate()).padStart(2, '0');
        return {
            year,
            month,
            date,
            dash: `${year}-${month}-${date}`,
            key: `${year}${month}${date}`
        }
    }//date_to_information

    /**
     * yyyy-mm-ddT00:00:00 형식의 문자열을 받아 Date 객체로 반환합니다.
     * @param {String} str 
     * @memo 아이폰 7미만 대응을 위함
     */
    static string_to_date(str) {
        if (!str) { throw new Error("no String for date parser"); }
        const { year, month, date, hour, minute, second } = string_to_object(str);
        return new Date(year, month - 1, date, hour, minute, second);
    }//string_to_date

    /**
     * 
     * @param {*} str 
     * @returns {Object} values are integar
     * @memo month is strictly transformed to Integar. if you want to use as 'Date' Keep in mind that you have to subtract 1 from month value
     */
    static string_to_object(str) {
        if (!str) { throw new Error("no string value for Date String Parser") }
        if (str.includes("T")) {
            const [yyyymmdd, hhmmss] = str.split("T");
            const [yyyy, mm, dd] = yyyymmdd.split("-").map(s => parseInt(s));
            const [h, m, s] = hhmmss.split(":").map(s => parseInt(s));
            return {
                year: yyyy,
                month: mm,
                date: dd,
                hour: h,
                minute: m,
                second: s
            };
        } else {
            const [yyyy, mm, dd] = str.split("-").map(s => parseInt(s));
            return {
                year: yyyy,
                month: mm,
                date: dd,
                hour: 0,
                minute: 0,
                second: 0
            };
        }
    }//string_to_object

    /**
     * date로 부터 getDay 해온 정수값을 
     * 한글 요일 / 영어 요일로 반환
     * @param {} date Date 객체 (new Date)
     */
    static get_day_kr_eng(date) {
        if (!date) { throw new Error("Date value is needed"); }
        const day = date.getDay();
        let kr = null;
        let eng = null;
        let engShort = null;
        switch (day) {
            case 0: {
                kr = "일";
                eng = "sunday";
                engShort = "sun";
            } break;
            case 1: {
                kr = "월";
                eng = "monday";
                engShort = "mon";
            } break;
            case 2: {
                kr = "화";
                eng = "tuesday";
                engShort = "tue";
            } break;
            case 3: {
                kr = "수";
                eng = "wednesday";
                engShort = "wed";
            } break;
            case 4: {
                kr = "목";
                eng = "thursday";
                engShort = "thu";
            } break;
            case 5: {
                kr = "금";
                eng = "friday";
                engShort = "fri";
            } break;
            case 6: {
                kr = "토";
                eng = "saturday";
                engShort = "sat";
            } break;
        }
        return { kr, eng, engShort };
    }//get_day_kr_eng

    /**
     * 서버타임 가져오기
     * @returns 
     */
    static async get_server_time() {
        try {
            const response = await fetch(window.location.href, { method: 'HEAD' });
            const serverDateString = response.headers.get('Date'); // 서버의 GMT 시간
            const serverTime = new Date(serverDateString);
            // console.log("서버 시간:", serverTime);
            return serverTime;
        } catch (e) {
            return new Date();
        }
    }//get_server_time

}//DateParser

/**
 * 캘린더 일정 가져오는 방법
 */
export class ScheduleModel {
    static get id_timetable() {
        return "mntmtbl";
    }
    static get id_calendar() {
        return "mncalndr";
    }

    static async fetch_json() {
        /* DEV */
        const URL = "/governor/json/components/sampleCalendar.json";
        /* 발행 */
        // const URL = "/governor/lib/components/realCalendar.txt";
        const dataJson = await fetch(URL).then(res => res.text());
        const dataSample = JSON.parse(dataJson);
        return dataSample;
    }

    /**
     * 일간 데이터 가져오기
     * @returns {Boolean | Object}
     */
    static async get_daily_timetable({ year, month, date }) {
        const dataSample = await ScheduleModel.fetch_json();
        const key = `${year}${month}${date}`;
        /* .... 비동기로 받아옵니다 */
        if (dataSample[key]) {
            return { [key]: dataSample[key] };
        } else {
            return false;
        }
    }//get_daily_timetable

    /**
     * 월간 데이터 가져오기
     * @returns {Boolean | Object}
     */
    static async get_monthly_timetable({ year, month }) {
        const dataSample = await ScheduleModel.fetch_json();
        const keys = Object.keys(dataSample);
        const filtered = keys.filter(key => key.includes(`${year}${month}`));
        if (!filtered.length) {
            return false;
        }
        const result = {};
        filtered.forEach(key => {
            result[key] = dataSample[key];
        });
        return result;
    }//get_daily_timetable
}//class:ScheduleModel

    static fill_timetable({ targetId, timetables, year, month, date }) {
        const $wrap = document.getElementById(targetId);
        $wrap.innerHTML = "";

        /* early return(1) */
        if (!$wrap) {
            throw new Error("id 로부터 DOM 찾을 수 없음");
            return;
        }

        /* 제목 영역 */
        const todayYear = String(new Date().getFullYear());
        const $areaTitle = MainPageView.main_article_title({
            type: "timetable",
            icInfo: {
                url: "/governor/media/img/icon/ic_calendar_chk.svg",
                alt: "아이콘:달력"
            },
            title: `${String(year) !== todayYear ? year + "년 " : ""}${month}월 ${date}일 일정안내`
        });
        $wrap.appendChild($areaTitle);

        /* class */
        const clss = "timetable";

        /* early return(2) */
        if (!timetables) {
            DomCommonView.add_p_for_wrap({
                $target: $wrap,
                content: ["공개된 일정이 없습니다."],
                clss: `${clss}-message-empty`
            });
            return;
        }

        const dataTable = Object.values(timetables)?.[0];
        dataTable.forEach(item => {
            const { hour, minute, title, location, isLive, url } = item;
            const $li = document.createElement("LI");
            $li.classList.add(`${clss}-item`);
            $wrap.appendChild($li);

            const $frag = url ? document.createElement("A") : document.createDocumentFragment();
            if (url) {
                $frag.classList.add(`${clss}-item-link`);
                $frag.href = url;
                $frag.target = "_blank";
            }

            DomCommonView.add_p_for_wrap({
                $target: $frag,
                content: [`${String(hour).padStart(2, "0")}:${String(minute).padStart(2, "0")}`],
                clss: `${clss}-time`
            });

            const $info = document.createElement("DIV");
            $info.classList.add(`${clss}-info`);
            $frag.appendChild($info);

            const $title = document.createElement("P");
            $title.classList.add(`${clss}-title`);
            $title.innerHTML = title;
            $info.appendChild($title);

            if (isLive) {
                DomCommonView.add_p_for_wrap({
                    $target: $title,
                    content: ["LIVE"],
                    clssList: [`${clss}-badge`, `${clss}-badge-live`],
                    tagAlt: "SPAN"
                });
            }

            const $loc = document.createElement("P");
            $loc.classList.add(`${clss}-location`);
            $loc.textContent = location;
            // $loc.prepend(DomCommonView.make_tabler_icon(["ti", "ti-map-pin"]));
            $loc.prepend(DomCommonView.make_icon_map_pin());
            $info.appendChild($loc);

            /* 최종 */
            $li.appendChild($frag);
        });
    }//fill_timetable

    /**
     * 달력 채우기
     * - month String으로 받아옴 / 5월이면 "05"
     * @memo 2026-11-17 웹접근성 심사로 인해 TABLE 방식으로 변경
     * @param {} 
     */
    static fill_calendar({ targetId, timetables, year, month }) {
        const $wrap = document.getElementById(targetId);
        $wrap.innerHTML = "";
        $wrap.dataset.dateRough = `${year}-${month}-01`;

        if (!$wrap) {
            throw new Error("id 로부터 DOM 찾을 수 없음");
        }
        const clss = "calendar";

        /* 상단 TOP */
        const $top = document.createElement("DIV");
        $top.classList.add(`${clss}-top`);
        $wrap.appendChild($top);

        const make_btn = (dir) => {
            const $btn = document.createElement("BUTTON");
            $btn.type = "button";
            $btn.classList.add(`${clss}-btn-month`);
            $btn.classList.add(dir);
            $btn.dataset.direction = dir;
            $btn.textContent = dir === "prev" ? "‹" : "›";
            $btn.setAttribute("aria-label",dir==="prev" ? "이전달" : "다음달");
            return $btn;
        }
        $top.appendChild(make_btn("prev"));
        DomCommonView.add_p_for_wrap({
            $target: $top,
            content: [`${year}.${month}`],
            clss: `${clss}-top-info`,
            tagAlt: "H4"
        });
        $top.appendChild(make_btn("next"));

        /* ---------------------------------- */
        /* 달력 부분 > TABLE */
        const $table = document.createElement("TABLE");
        $table.classList.add(`${clss}-tbl`);
        $wrap.appendChild($table);

        DomCommonView.add_p_for_wrap({
            $target: $table,
            content: [`${year}년 ${month}월 달력. 일,월,화,수,목,금,토에 해당하는 각 일의 정보를 담고있습니다.`],
            clss: `hidden-readable`,
            tagAlt: "CAPTION"
        });

        /* 달력 부분 */
        const $thead = document.createElement("THEAD");
        $thead.classList.add(`${clss}-days`);
        $table.appendChild($thead);

        const $theadTr = document.createElement("TR");
        $thead.appendChild($theadTr);

        DomCommonView.add_p_for_wrap({
            $target: $theadTr,
            content: ["일", "월", "화", "수", "목", "금", "토"],
            clss: `${clss}-days-day`,
            tagAlt: "TH"
        });
        /* ---------------------------------- */
        /* 달력:일 채우기 */
        const $tbody = document.createElement("TBODY");
        $tbody.classList.add(`${clss}-dates`);
        $table.appendChild($tbody);

        const firstDay = new Date(year, Number(month) - 1, 1).getDay();
        const lastDate = new Date(year, Number(month), 0).getDate();
        const prevLastDate = new Date(year, Number(month) - 1, 0).getDate();

        /** 
         * 일 버튼 만들기 
         * */
        const make_date = ({ year, month, date, day, active = false, isToday = false, hasEvent, selected }) => {
            /* TD */
            const $td = document.createElement("TD");
            $td.classList.add(`${clss}-td`);
            /* BUTTON */
            const $dateBtn = document.createElement(active ? "BUTTON" : "DIV");
            $dateBtn.classList.add(`${clss}-date`);
            if (isToday) {
                $dateBtn.classList.add(`today`);
            }
            if (selected) {
                $dateBtn.classList.add("selected");
                $dateBtn.title = "선택됨";
            }
            if (hasEvent) {
                $dateBtn.classList.add(`hasEvent`);
            }
            $dateBtn.setAttribute("aria-label", `${year}년 ${month}월 ${date}일 ${MainScheduleView.get_day_kr(day)}요일${isToday ? "(오늘)" : ""}, 일정 ${hasEvent ? "있음" : "없음"}`)
            $td.appendChild($dateBtn);
            if (active) {
                $dateBtn.classList.add(`${clss}-btn-date`);
                $dateBtn.dataset.date = DateParser.date_to_string(new Date(year, month, date)).dash;
                $dateBtn.type = "button";
            } else {
                $dateBtn.classList.add("othermonth");
            }//ifElse
            DomCommonView.add_p_for_wrap({
                $target: $dateBtn,
                content: [date],
                clss: `${clss}-btn-date-txt`,
                tagAlt: "SPAN"
            });
            return $td;
        }//make_date

        /* TBODY TR */
        let $tbdyTR = document.createElement("TR");
        $tbody.appendChild($tbdyTR);

        let cellCount = 0;

        /* 이전달 */
        for (let i = firstDay - 1; i >= 0; i--) {
            cellCount++;
            $tbdyTR.appendChild(make_date({
                year,
                month: Number(month) - 1,
                date: prevLastDate - i,
                day: cellCount - 1
            }));
            if (cellCount >= 7) {
                cellCount = 0;
                $tbody.appendChild($tbdyTR);
                $tbdyTR = document.createElement("TR");
            }
        }//이전달

        /* 현재달 */
        const today = DateParser.date_to_string(new Date());
        const remember = $wrap.dataset.dateRemember;
        if (!remember) {
            throw new Error("data-date-remember 값 없음");
        }

        for (let i = 1; i <= lastDate; i++) {
            cellCount++;

            const dayString = String(i).padStart(2, '0');
            const dateStr = `${year}-${month}-${dayString}`;
            const dateKey = `${year}${month}${dayString}`;
            const isToday = (year === today.year && month === today.month && dayString === today.date);
            const hasEvent = timetables[dateKey];
            const $dateTD = make_date({
                year,
                month: Number(month) - 1,
                date: i,
                day: cellCount - 1,
                active: true,
                isToday,
                hasEvent,
                selected : dateStr === remember
            });
            
            $tbdyTR.appendChild($dateTD);

            if (cellCount >= 7) {
                cellCount = 0;
                $tbody.appendChild($tbdyTR);
                $tbdyTR = document.createElement("TR");
            }
        }//현재달

        /* 다음달 */
        const totalCells = firstDay + lastDate;
        const remainingCells = totalCells % 7 === 0 ? 0 : 7 - (totalCells % 7);
        for (let i = 1; i <= remainingCells; i++) {
            cellCount++;
            $tbdyTR.appendChild(make_date({
                year,
                month: Number(month),
                date: i,
                active: false,
                day: cellCount - 1
            }));
            if (cellCount >= 7) {
                cellCount = 0;
                $tbody.appendChild($tbdyTR);
                $tbdyTR = document.createElement("TR");
            }
        }
    }//fill_calendar

    static get_day_kr(day) {
        switch (day) {
            case 0: { return "일" } break;
            case 1: { return "월" } break;
            case 2: { return "화" } break;
            case 3: { return "수" } break;
            case 4: { return "목" } break;
            case 5: { return "금" } break;
            case 6: { return "토" } break;
            default: {
                return "ERROR"
            } break;
        }
    }//get_day_kr
}//MainScheduleView
